const views = ['routesView', 'shopsView', 'productsView']; let history = ['routesView']; function showView(viewId, title, showBack) { views.forEach(v => { let el = document.getElementById(v); el.classList.remove('active'); }); let activeEl = document.getElementById(viewId); activeEl.classList.add('active'); document.getElementById('headerTitle').innerText = title; document.getElementById('backBtn').style.display = showBack ? 'block' : 'none'; if(viewId !== 'productsView') { document.getElementById('bottomBar').style.display = 'none'; } else { updateCartTotal(); } } function goBack() { history.pop(); let prev = history[history.length - 1]; if (prev === 'routesView') { showView(prev, 'Sales Routes', false); } else if (prev === 'shopsView') { showView(prev, currentRoute, true); } } function openRoute(routeName) { currentRoute = routeName; history.push('shopsView'); renderShopsList(); showView('shopsView', routeName, true); } function renderShopsList() { let list = shopsDB[currentRoute] || []; let html = ''; list.forEach((shop, index) => { html += `
${index + 1}. ${shop.name}
Tap to select shop & add products
`; }); document.getElementById('shopsList').innerHTML = html; } function openAddShopModal() { document.getElementById('newShopName').value = ''; let select = document.getElementById('insertAfterShop'); let html = ''; let list = shopsDB[currentRoute] || []; list.forEach((shop, index) => { let selected = (index === list.length - 1) ? 'selected' : ''; html += ``; }); select.innerHTML = html; document.getElementById('addShopModal').style.display = 'flex'; } function closeAddShopModal() { document.getElementById('addShopModal').style.display = 'none'; } function saveNewShop() { let name = document.getElementById('newShopName').value.trim(); if(!name) { alert("Dukan ka naam likhna zaroori hai!"); return; } let index = parseInt(document.getElementById('insertAfterShop').value); let newShop = { id: 'new_' + Date.now(), name: name }; if(index === -1) { shopsDB[currentRoute].unshift(newShop); } else { shopsDB[currentRoute].splice(index + 1, 0, newShop); } localStorage.setItem('shopsDB', JSON.stringify(shopsDB)); closeAddShopModal(); renderShopsList(); } function openShop(shopId, shopName) { currentShop = shopName; currentShopId = shopId; cart = {}; history.push('productsView'); renderShopHeader(); renderProducts(); showView('productsView', 'Order', true); } function renderShopHeader() { let shop = shopsDB[currentRoute].find(s => s.id === currentShopId); let locIconClass = (shop.gpsSaved || shop.url) ? 'active-gps' : ''; let html = `
${shop.name}
${currentRoute}
`; document.getElementById('shopHeaderContainer').innerHTML = html; } function saveGPS(route, shopId) { if ("geolocation" in navigator) { navigator.geolocation.getCurrentPosition(function(position) { let shop = shopsDB[route].find(s => s.id === shopId); shop.lat = position.coords.latitude; shop.lng = position.coords.longitude; shop.gpsSaved = true; localStorage.setItem('shopsDB', JSON.stringify(shopsDB)); alert(shop.name + " ki location save ho gayi!"); renderShopHeader(); }, function(error) { alert("GPS error. Location on kijiye aur permission dijiye."); }); } else { alert("Aapka device GPS support nahi karta."); } } function navigateShop(route, shopId) { let shop = shopsDB[route].find(s => s.id === shopId); if(shop.url) { window.open(shop.url, '_blank'); } else if (shop.lat && shop.lng) { window.open(`https://www.google.com/maps/search/?api=1&query=${shop.lat},${shop.lng}`, '_blank'); } else { alert("Pehle 📍 button daba kar location save karein."); } } function renderProducts() { let html = ''; productsDB.forEach(p => { let schemeText = p.bandha ? `Min 1 bandha (${p.bandha} pcs)` : ""; if(p.scheme) schemeText += (schemeText ? " | " : "") + p.scheme; let qty = cart[p.id]?.qty || 0; let currentRateText = calculateApplicableRate(p, qty); // UI me agar Rate aur MRP same hain, toh "Regular rate" dikhega let displayRateUI = (currentRateText == p.mrp) ? "Regular rate" : "₹" + currentRateText; html += `
${p.name}
${schemeText ? `
${schemeText}
` : ''}
MRP: ₹${p.mrp}
Rate: ${displayRateUI}
`; }); document.getElementById('productsList').innerHTML = html; } function calculateApplicableRate(product, qty) { let finalRate = product.rate; if(product.slabs) { let sortedSlabs = [...product.slabs].sort((a,b) => b.min - a.min); for(let slab of sortedSlabs) { if(qty >= slab.min) { finalRate = slab.rate; break; } } } return finalRate; } function updateQty(pId, change) { let p = productsDB.find(x => x.id === pId); let input = document.getElementById(`qty_${pId}`); let current = parseInt(input.value) || 0; let step = p.bandha ? p.bandha : 1; let newQty = current + (change * step); if (newQty < 0) newQty = 0; input.value = newQty; processCart(p, newQty); } function manualQty(pId) { let p = productsDB.find(x => x.id === pId); let input = document.getElementById(`qty_${pId}`); let newQty = parseInt(input.value) || 0; if(newQty < 0) newQty = 0; processCart(p, newQty); } function processCart(product, qty) { let actualRate = calculateApplicableRate(product, qty); let displayRateUI = (actualRate == product.mrp) ? "Regular rate" : "₹" + actualRate; document.getElementById(`rate_${product.id}`).innerText = displayRateUI; if (qty > 0) { cart[product.id] = { p: product, qty: qty, actualRate: actualRate }; } else { delete cart[product.id]; } updateCartTotal(); } function updateCartTotal() { let totalAmt = 0; let totalItems = 0; for (let key in cart) { totalAmt += (cart[key].actualRate * cart[key].qty); totalItems++; } document.getElementById('totalVal').innerText = totalAmt.toFixed(2); document.getElementById('totalItems').innerText = totalItems; if(totalItems > 0) { document.getElementById('bottomBar').style.display = 'flex'; } else { document.getElementById('bottomBar').style.display = 'none'; } } function generateWhatsApp() { if(Object.keys(cart).length === 0) return; let msg = `*${currentShop}*\n*${currentRoute}*\n\n`; for (let key in cart) { let item = cart[key]; let p = item.p; let qtyText = `${item.qty} pcs`; if (p.bandha && item.qty >= p.bandha) { let bandhaCount = Math.floor(item.qty / p.bandha); let loosePcs = item.qty % p.bandha; if (loosePcs === 0) { qtyText = `${bandhaCount} Bandha (${item.qty} pcs)`; } else { qtyText = `${bandhaCount} Bandha & ${loosePcs} pcs (${item.qty} pcs)`; } } let displayRate = (item.actualRate == p.mrp) ? "Regular rate" : item.actualRate; // 'Item:' prefix hata diya gaya hai msg += `${p.name}\nMrp: ${p.mrp}\nRate: ${displayRate}\nQuantity: ${qtyText}\n\n`; } let encodedMsg = encodeURIComponent(msg); window.open(`https://wa.me/?text=${encodedMsg}`, '_blank'); } function filterShops() { let input = document.getElementById('shopSearch').value.toLowerCase(); let items = document.getElementsByClassName('shop-item'); for (let i = 0; i < items.length; i++) { let txtValue = items[i].textContent || items[i].innerText; items[i].style.display = txtValue.toLowerCase().indexOf(input) > -1 ? "" : "none"; } } function filterProducts() { let input = document.getElementById('productSearch').value.toLowerCase(); let items = document.getElementsByClassName('product-item'); for (let i = 0; i < items.length; i++) { let txtValue = items[i].textContent || items[i].innerText; items[i].style.display = txtValue.toLowerCase().indexOf(input) > -1 ? "" : "none"; } }
